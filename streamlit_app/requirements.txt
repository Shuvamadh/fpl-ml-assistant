# Pinned deliberately. Unpinned deps mean the environment gets rebuilt from
# latest on every redeploy, so a LightGBM major bump can break models/*.txt
# loading ("Model format error") on a redeploy that changed nothing in this
# repo. Bump these on purpose, not by accident.
#
# These versions target Python 3.11, which is what the Dockerfile pins
# (`FROM python:3.11-slim`) for the Render/Docker deployment. Do not bump
# pandas/numpy past what has cp311 wheels without also bumping the
# Dockerfile's Python version -- numpy>=2.5 requires Python>=3.12 and broke
# the Render build when paired with 3.11.

pandas==2.2.3
numpy==2.1.3
requests==2.32.3
lightgbm==4.7.0
